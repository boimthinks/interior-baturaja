## 7. Konfigurasi Astro JS Content Collection & Schema Frontmatter

### A. File `src/content/config.ts`
```typescript
import { defineCollection, z } from 'astro:content';

const postsCollection = defineCollection({
  type: 'content',
  schema: z.object({
    title: z.string(),
    description: z.string(),
    pubDate: z.coerce.date(),
    updatedDate: z.coerce.date().optional(),
    category: z.enum(['Blog', 'Portfolio']),
    tags: z.array(
      z.enum([
        'Kitchen Set',
        'Lemari & Kamar Set',
        'Backdrop TV & Partisi',
        'Interior Kantor & Instansi',
        'Interior Toko, Mall & Booth',
        'Edukasi & Material',
        'Harga & RAB',
        'Baturaja',
        'OKU Timur',
        'OKU Selatan',
        'Muara Enim',
        'Prabumulih',
      ])
    ),
    targetKeywords: z.array(z.string()),
    relatedServiceUrl: z.string(),
    // Wajib diisi jika category === 'Portfolio'
    projectDetails: z
      .object({
        clientOrProject: z.string(),
        location: z.string(),
        material: z.string(),
        duration: z.string(),
        year: z.number(),
      })
      .optional(),
    // Wajib ada (minimal 3 FAQ) di setiap post untuk GEO & JSON-LD FAQPage
    faqs: z
      .array(
        z.object({
          question: z.string(),
          answer: z.string(),
        })
      )
      .min(3),
  }),
});

export const collections = {
  posts: postsCollection,
};
```
